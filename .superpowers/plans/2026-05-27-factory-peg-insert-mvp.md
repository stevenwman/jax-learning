### Task 0.2: USD mesh-only extract (peg + hole)

**Phase 0a goal:** Pull `factory_peg_8mm.usd` + `factory_hole_8mm.usd` from NVIDIA's public S3 mirror. Walk each USD stage with `pxr.Usd` and extract the triangle meshes directly to OBJ. CoACD-decomp the hole (non-convex annulus). Harvest USD metadata (positions, scales, mass if present) and dump as JSON for cross-reference against IsaacLab's `factory_tasks_cfg.py` values. No URDF middleman.

**Why mesh-only (not the URDF pipeline):** The only thing we actually need from the USDs is the triangle mesh geometry — peg+hole are single rigid bodies with no joints, so URDF link/joint structure adds no value. Mass and inertia are simple analytical values (peg = cylinder 19g; hole = static base, no inertia needed) authored directly in our MJCF. Skipping URDF cuts the dependency on `nvidia-srl-usd-to-urdf`, eliminates a tool-discovery + URDF-cleanup cycle, and reduces failure modes. We still get Isaac's exact geometry — that's the parity payoff.

**Why harvest USD metadata:** Cross-checking `factory_peg_8mm.usd`'s reported positions/scales/masses against IsaacLab's `factory_tasks_cfg.py:Peg8mm` (diameter=7.986mm, height=50mm, mass=19g) catches asset drift (Isaac updates a USD without bumping the config, or vice versa). The harvested JSON gets committed alongside the OBJs so future agents can audit parity.

- [ ] **Step 2: Write the extract pipeline script**

Create `scripts/factory/extract_usds.py`:

```python
"""Download Factory USDs from NVIDIA's public S3 mirror, extract triangle meshes
+ harvest metadata. No URDF middleman.

Phase 0a of the Factory MJX/Warp port: peg + hole only. Phase 4 (deferred)
extends to nut + bolt + gear.

Outputs (relative to repo root):
  .tmp/factory_usds/factory_peg_8mm.usd        (cache, gitignored)
  .tmp/factory_usds/factory_hole_8mm.usd       (cache, gitignored)
  jax_rl/envs/manipulation/factory/assets/peg_insert/extracted/
    ├── peg_8mm.obj                 (extracted triangle mesh, convex)
    ├── hole_8mm_raw.obj            (extracted triangle mesh, non-convex)
    ├── hole_decomp/hole_*.obj      (CoACD pieces, ≤32)
    └── metadata.json               (USD positions, scales, masses for parity audit)

Usage:
  uv run python scripts/factory/extract_usds.py
"""
import json
import subprocess
import urllib.request
from pathlib import Path

import trimesh
from pxr import Usd, UsdGeom, UsdPhysics

REPO_ROOT = Path(__file__).resolve().parents[2]
USD_CACHE = REPO_ROOT / ".tmp" / "factory_usds"
EXTRACT_ROOT = (REPO_ROOT / "jax_rl" / "envs" / "manipulation" / "factory" /
                "assets" / "peg_insert" / "extracted")
HOLE_DECOMP = EXTRACT_ROOT / "hole_decomp"

S3_BASE = ("https://omniverse-content-production.s3-us-west-2.amazonaws.com"
           "/Assets/Isaac/4.2/Isaac/IsaacLab/Factory")
ASSETS = {
    "factory_peg_8mm.usd":  "peg_8mm",
    "factory_hole_8mm.usd": "hole_8mm_raw",
}


def download(name: str) -> Path:
    dst = USD_CACHE / name
    if dst.exists() and dst.stat().st_size > 0:
        print(f"  cached: {dst} ({dst.stat().st_size} bytes)")
        return dst
    USD_CACHE.mkdir(parents=True, exist_ok=True)
    url = f"{S3_BASE}/{name}"
    print(f"  downloading: {url}")
    urllib.request.urlretrieve(url, dst)
    print(f"  → {dst} ({dst.stat().st_size} bytes)")
    return dst


def triangulate(indices, counts):
    """Convert USD's variable-N face-vertex-counts to triangle indices.

    USD meshes may have quads or higher n-gons; trimesh + MuJoCo want triangles.
    Fan-triangulation: for an n-gon (v0, v1, ..., vn-1), emit (v0, v1, v2),
    (v0, v2, v3), ..., (v0, vn-2, vn-1). Assumes convex polygons.
    """
    tris = []
    cursor = 0
    for n in counts:
        for k in range(1, n - 1):
            tris.append([indices[cursor], indices[cursor + k], indices[cursor + k + 1]])
        cursor += n
    return tris


def extract_meshes_and_metadata(usd_path: Path, out_stem: str) -> dict:
    """Walk USD stage, extract all UsdGeomMesh prims as a single combined OBJ.

    Returns dict of metadata (mesh stats, prim transforms, mass if present).
    """
    stage = Usd.Stage.Open(usd_path.as_posix())
    if stage is None:
        raise RuntimeError(f"Failed to open USD: {usd_path}")

    all_verts = []
    all_tris = []
    vert_offset = 0
    prim_meta = []

    for prim in stage.Traverse():
        mesh = UsdGeom.Mesh(prim)
        if not mesh:
            continue
        points = mesh.GetPointsAttr().Get()
        indices = mesh.GetFaceVertexIndicesAttr().Get()
        counts = mesh.GetFaceVertexCountsAttr().Get()
        if points is None or indices is None or counts is None:
            continue
        verts = [(p[0], p[1], p[2]) for p in points]
        tris = triangulate(list(indices), list(counts))
        # Re-offset triangle indices into the combined vertex list.
        all_verts.extend(verts)
        all_tris.extend([(t[0] + vert_offset, t[1] + vert_offset, t[2] + vert_offset)
                          for t in tris])
        prim_meta.append({
            "prim_path": str(prim.GetPath()),
            "n_verts": len(verts),
            "n_tris": len(tris),
        })
        vert_offset += len(verts)

    if not all_verts:
        raise RuntimeError(f"No UsdGeomMesh prims found in {usd_path}")

    # Harvest physics + transform metadata for the prim hierarchy.
    metadata = {
        "source_usd": usd_path.name,
        "out_stem": out_stem,
        "n_verts_total": len(all_verts),
        "n_tris_total": len(all_tris),
        "prims": prim_meta,
        "stage_metadata": {
            "default_prim": str(stage.GetDefaultPrim().GetPath()) if stage.GetDefaultPrim() else None,
            "up_axis": UsdGeom.GetStageUpAxis(stage),
            "meters_per_unit": UsdGeom.GetStageMetersPerUnit(stage),
        },
        "physics": _harvest_physics(stage),
    }

    # Write the combined OBJ.
    EXTRACT_ROOT.mkdir(parents=True, exist_ok=True)
    obj_path = EXTRACT_ROOT / f"{out_stem}.obj"
    mesh = trimesh.Trimesh(vertices=all_verts, faces=all_tris)
    mesh.export(obj_path.as_posix())
    print(f"  {usd_path.name} → {obj_path} ({len(all_verts)} verts, {len(all_tris)} tris)")
    return metadata


def _harvest_physics(stage):
    """Pull mass + collision + rigid-body schemas off any prim that has them."""
    rows = []
    for prim in stage.Traverse():
        mass_api = UsdPhysics.MassAPI(prim) if prim.HasAPI(UsdPhysics.MassAPI) else None
        rigid_api = UsdPhysics.RigidBodyAPI(prim) if prim.HasAPI(UsdPhysics.RigidBodyAPI) else None
        col_api = UsdPhysics.CollisionAPI(prim) if prim.HasAPI(UsdPhysics.CollisionAPI) else None
        if mass_api or rigid_api or col_api:
            rows.append({
                "prim_path": str(prim.GetPath()),
                "mass": float(mass_api.GetMassAttr().Get()) if mass_api and mass_api.GetMassAttr().IsValid() else None,
                "rigid_body": rigid_api is not None,
                "collision": col_api is not None,
            })
    return rows


def coacd_decomp(obj_path: Path, out_dir: Path, threshold: float = 0.05) -> list:
    out_dir.mkdir(parents=True, exist_ok=True)
    out_template = out_dir / (obj_path.stem.replace("_raw", "") + ".obj")
    cmd = ["coacd", obj_path.as_posix(), "-o", out_template.as_posix(),
           "--threshold", str(threshold)]
    subprocess.run(cmd, check=True)
    return sorted(out_dir.glob(f"{obj_path.stem.replace('_raw', '')}*.obj"))


def main():
    print("== Phase 0a: USD mesh-only extract (peg + hole) ==\n")

    print("Step 1/3: Download USDs from S3")
    usd_paths = {name: download(name) for name in ASSETS}

    print("\nStep 2/3: Extract triangle meshes + harvest metadata")
    all_metadata = {}
    for usd_name, out_stem in ASSETS.items():
        all_metadata[usd_name] = extract_meshes_and_metadata(usd_paths[usd_name], out_stem)

    # Write metadata.json for parity audit.
    metadata_path = EXTRACT_ROOT / "metadata.json"
    metadata_path.write_text(json.dumps(all_metadata, indent=2))
    print(f"\n  metadata → {metadata_path}")
    print("  Cross-check these values against IsaacLab factory_tasks_cfg.py:")
    print("    Peg8mm:  diameter=7.986mm, height=50mm, mass=19g")
    print("    Hole8mm: diameter=8.1mm (inner), height=25mm")

    print("\nStep 3/3: CoACD decomp on hole (non-convex)")
    hole_raw = EXTRACT_ROOT / "hole_8mm_raw.obj"
    pieces = coacd_decomp(hole_raw, HOLE_DECOMP)
    print(f"  hole decomposed into {len(pieces)} convex pieces")
    if len(pieces) > 32:
        print("  WARNING: >32 pieces. Bump CoACD --threshold (e.g., 0.05 → 0.08).")

    print("\n== Phase 0a: DONE ==")
    print(f"Meshes at: {EXTRACT_ROOT}")
    print(f"Hole decomp: {HOLE_DECOMP}")


if __name__ == "__main__":
    main()
```

Runs three tests:
  1. straight drop  — peg at (0, 0, 80mm), zero vel, identity orient
  2. lateral nudge  — peg at (2mm, 0, 80mm)
  3. tilt           — peg at (0, 0, 80mm), 10° tilt around x-axis

Pass criteria per test:
  - No NaN in qpos/qvel after 500 substeps × 2ms = 1s sim
  - Final z ∈ [-1mm, +1mm] (settled, NOT tunneled)
  - Final lateral ‖xy‖ < 2mm
  - Contact count > 0 sustained for >50 steps

Usage:
  uv run python jax_rl/envs/manipulation/factory/assets/peg_insert/spike_drop.py

import jax
import jax.numpy as jp
import mujoco
from mujoco import mjx
import numpy as np

HERE = Path(__file__).parent
EXTRACTED = HERE / "extracted"
HOLE_DECOMP = EXTRACTED / "hole_decomp"
PEG_MESH = EXTRACTED / "peg_8mm.obj"


def build_mjcf() -> str:
    """Read spike.xml and inject peg + hole mesh refs from extracted/."""
    if not PEG_MESH.exists():
        raise FileNotFoundError(f"Missing {PEG_MESH}. "
                                "Run scripts/factory/extract_usds.py first (Task 0.2).")
    hole_pieces = sorted(HOLE_DECOMP.glob("hole*.obj"))
    if not hole_pieces:
        raise FileNotFoundError(f"No hole decomp pieces in {HOLE_DECOMP}. "
                                "Task 0.2 should have produced them.")
    # Mesh paths in MJCF are relative to compiler `meshdir="extracted"`.
    # Peg is a single mesh at `extracted/peg_8mm.obj` (referenced as `peg_8mm.obj`).
    # Hole pieces at `extracted/hole_decomp/hole_*.obj`.
    assets_block = "\n".join(
        [f'    <mesh name="peg" file="{PEG_MESH.name}"/>']
        + [f'    <mesh name="hole_{i:03d}" file="hole_decomp/{p.name}"/>'
           for i, p in enumerate(hole_pieces)]
    )
    peg_geom = (
        '      <geom name="peg" type="mesh" mesh="peg" '
        'condim="6" friction="1.0 0.01 0.0001" rgba="0.2 0.6 0.9 1"/>'
    )
    hole_geoms = "\n".join(
        f'      <geom name="hole_{i:03d}" type="mesh" mesh="hole_{i:03d}" '
        f'condim="6" friction="1.0 0.01 0.0001" '
        f'solref="0.004 1" solimp="0.98 0.995 0.001"/>'
        for i, p in enumerate(hole_pieces)
    )
    template = (HERE / "spike.xml").read_text()
    return (template
            .replace("<!-- decomp pieces inserted at load time -->", hole_geoms)
            .replace("<!-- peg geom inserted at load time (mesh from extracted/peg_8mm.obj) -->",
                     peg_geom)
            .replace("<!-- Filenames are filled in by spike_drop.py at load time -->",
                     assets_block))


def run_test(name: str, init_qpos: np.ndarray, init_qvel: np.ndarray) -> dict:
    # Mesh files must be passed as assets bytes so MuJoCo's compiler can find them.
    assets = {PEG_MESH.name: PEG_MESH.read_bytes()}
    for p in sorted(HOLE_DECOMP.glob("hole*.obj")):
        assets[f"hole_decomp/{p.name}"] = p.read_bytes()
    mj_model = mujoco.MjModel.from_xml_string(build_mjcf(), assets=assets)
    mj_model.opt.timestep = 0.002
    mjx_model = mjx.put_model(mj_model, impl="warp")
    mjx_data = mjx.make_data(mjx_model)
    mjx_data = mjx_data.replace(
        qpos=jp.array(init_qpos),
        qvel=jp.array(init_qvel),
    )

    @jax.jit
    def step(data):
        return mjx.step(mjx_model, data)

    contact_counts = []
    for i in range(500):
        mjx_data = step(mjx_data)
        contact_counts.append(int(jp.sum(mjx_data.contact.dist < 0)))

    final_qpos = np.asarray(mjx_data.qpos)
    final_qvel = np.asarray(mjx_data.qvel)
    nan_in_state = bool(np.any(np.isnan(final_qpos)) or np.any(np.isnan(final_qvel)))
    final_z = final_qpos[2]
    final_xy = float(np.linalg.norm(final_qpos[:2]))
    sustained_contact = sum(1 for c in contact_counts[-100:] if c > 0)

    passed = (
        not nan_in_state
        and -0.001 <= final_z <= 0.001
        and final_xy < 0.002
        and sustained_contact > 50
    )
    print(f"  {'PASS' if passed else 'FAIL'} {name}: "
          f"z={final_z*1000:.2f}mm, ‖xy‖={final_xy*1000:.2f}mm, "
          f"sustained_contact={sustained_contact}, nan={nan_in_state}")
    return {"name": name, "passed": passed, "z": final_z, "xy": final_xy,
            "sustained_contact": sustained_contact, "nan": nan_in_state}


def main():
    # 1. Straight drop.
    qpos = np.array([0.0, 0.0, 0.080, 1.0, 0.0, 0.0, 0.0])  # freejoint: xyz + quat (w,x,y,z)
    qvel = np.zeros(6)
    r1 = run_test("straight_drop", qpos, qvel)

    # 2. Lateral nudge (2mm x offset).
    qpos = np.array([0.002, 0.0, 0.080, 1.0, 0.0, 0.0, 0.0])
    r2 = run_test("lateral_nudge", qpos, qvel)

    # 3. 10° tilt around x-axis.
    angle_rad = np.deg2rad(10.0)
    quat = np.array([np.cos(angle_rad/2), np.sin(angle_rad/2), 0, 0])
    qpos = np.array([0.0, 0.0, 0.080, *quat])
    r3 = run_test("tilt_10deg", qpos, qvel)

    print()
    if all(r["passed"] for r in (r1, r2, r3)):
        print("Phase 0 spike: PASS — Warp resolves peg-in-hole contact at 114µm clearance.")
        print("Green-light Phase 1.")
    else:
        print("Phase 0 spike: FAIL — see results above. Tune solver/decomp/friction before Phase 1.")
        print("Fallback cascade (in order):")
        print("  1. iterations 100 → 200, solref 0.004 → 0.002")
        print("  2. CoACD --threshold 0.05 → 0.02 (more pieces)")
        print("  3. condim=6 verified everywhere")
        print("  4. naccdmax bumped (set to 4000+ in MJCF option)")
        print("  5. widen clearance to 500µm (breaks parity, unblocks port)")

    main()

```python
"""EMA + denormalization + clipping for Factory pose-delta action.

Action is (6,) in [-1, 1]: [dx, dy, dz, drx, dry, drz].
Pipeline: raw → EMA-smooth → denorm to pose delta → clip relative to anchor.

State.info auto-reset trap: Brax-style auto-reset does NOT clear info between
episodes. EMA actions and prev_actions carry across resets if not explicitly
cleared. Use `reset_on_done` at the start of every env.step.
"""
import jax.numpy as jp


def apply_ema(raw: jp.ndarray, prev_ema: jp.ndarray, ema_factor: float = 0.2) -> jp.ndarray:
    """Smooth raw action via exponential moving average."""
    return ema_factor * raw + (1.0 - ema_factor) * prev_ema


def denormalize(
    ema: jp.ndarray,
    pos_threshold: jp.ndarray,
    rot_threshold: jp.ndarray,
    unidirectional_rot: bool = False,
) -> tuple[jp.ndarray, jp.ndarray]:
    """Convert EMA action [-1, 1]⁶ to pose deltas.

    If unidirectional_rot, yaw (rot[2]) is mapped from [-1, 1] to [-rot_thresh, 0]
    via `-(raw + 1) * 0.5 * rot_thresh`. This enforces tightening-only rotation
    for NutThread; PegInsert ignores it.
    """
    pos_delta = ema[:3] * pos_threshold
    rot_raw = ema[3:6]
    if unidirectional_rot:
        rot_yaw = -(rot_raw[2] + 1.0) * 0.5
        rot_raw = rot_raw.at[2].set(rot_yaw)
    rot_delta = rot_raw * rot_threshold
    return pos_delta, rot_delta


def clip_to_bounds(target: jp.ndarray, anchor: jp.ndarray, bounds: jp.ndarray) -> jp.ndarray:
    """Clip target so |target - anchor| ≤ bounds, elementwise.

    Used to keep the controller's commanded position close to the bolt/hole,
    preventing the policy from wandering during exploration.
    """
    delta = jp.clip(target - anchor, -bounds, bounds)
    return anchor + delta


def reset_on_done(ema: jp.ndarray, done: jp.ndarray) -> jp.ndarray:
    """Zero out EMA action on done=True per-env, preserve otherwise.

    ema: (N, 6) batched EMA action state stored in state.info.
    done: (N,) bool — True when episode just terminated.
    """

"""Factory reward terms — port of IsaacLab factory_utils.squashing_fn +
factory_env._get_factory_rew_dict.

All math is JIT-friendly JAX. Caller passes scalar pos/quat (already batched
by vmap if multi-env).
"""
from typing import Tuple

import jax.numpy as jp


def squashing_fn(x: jp.ndarray, a: float, b: float) -> jp.ndarray:
    """Smooth bump: 1 / (exp(a·x) + b + exp(-a·x)). Peaks at x=0 → 1/(2+b)."""
    return 1.0 / (jp.exp(a * x) + b + jp.exp(-a * x))


def keypoint_offsets(num_keypoints: int, scale: float) -> jp.ndarray:
    """Return (N, 3) offsets along z, evenly spaced in [-0.5, 0.5] * scale."""
    z = jp.linspace(-0.5, 0.5, num_keypoints) * scale
    out = jp.zeros((num_keypoints, 3))
    return out.at[:, 2].set(z)


def quat_rotate_vec(q: jp.ndarray, v: jp.ndarray) -> jp.ndarray:
    """Rotate vector v by quaternion q = (w, x, y, z)."""
    w, x, y, z = q[0], q[1], q[2], q[3]
    # Standard quat rotation: v' = q v q^*
    t = 2.0 * jp.cross(jp.array([x, y, z]), v)
    return v + w * t + jp.cross(jp.array([x, y, z]), t)


def keypoint_distance(
    held_pos: jp.ndarray, held_quat: jp.ndarray,
    target_pos: jp.ndarray, target_quat: jp.ndarray,
    offsets: jp.ndarray,
) -> jp.ndarray:
    """Mean L2 distance across N keypoints between held and target poses."""
    def transform(p, q, off):
        return p + quat_rotate_vec(q, off)
    held_kps   = jp.stack([transform(held_pos, held_quat, o) for o in offsets])
    target_kps = jp.stack([transform(target_pos, target_quat, o) for o in offsets])
    return jp.linalg.norm(held_kps - target_kps, axis=-1).mean()


def is_engaged(peg_z: float, hole_top_z: float, asset_height: float,
               engage_threshold: float) -> jp.ndarray:
    return jp.asarray(hole_top_z - peg_z > asset_height * engage_threshold)


def is_success(peg_z: float, hole_top_z: float, asset_height: float,
               success_threshold: float) -> jp.ndarray:
    return jp.asarray(hole_top_z - peg_z > asset_height * success_threshold)


def compute_reward(
    held_pos, held_quat, target_pos, target_quat,
    peg_z, hole_top_z,
    keypoint_coef_baseline: Tuple[float, float],
    keypoint_coef_coarse:   Tuple[float, float],
    keypoint_coef_fine:     Tuple[float, float],
    asset_height: float,
    engage_threshold: float,
    success_threshold: float,
    keypoint_scale: float,
    num_keypoints: int = 4,
) -> jp.ndarray:
    offsets = keypoint_offsets(num_keypoints, keypoint_scale)
    d = keypoint_distance(held_pos, held_quat, target_pos, target_quat, offsets)
    a0, b0 = keypoint_coef_baseline
    a1, b1 = keypoint_coef_coarse
    a2, b2 = keypoint_coef_fine
    r_baseline = squashing_fn(d, a0, b0)
    r_coarse   = squashing_fn(d, a1, b1)
    r_fine     = squashing_fn(d, a2, b2)
    r_engaged  = is_engaged(peg_z, hole_top_z, asset_height, engage_threshold).astype(jp.float32)
    r_success  = is_success(peg_z, hole_top_z, asset_height, success_threshold).astype(jp.float32)
    return r_baseline + r_coarse + r_fine + r_engaged + r_success
```

def _mat_to_quat(m: jp.ndarray) -> jp.ndarray:
    """3x3 rotation matrix → quat (w, x, y, z). JAX-stable 4-branch formula.

    Uses the standard algorithm: pick the largest diagonal-or-trace term to
    avoid sqrt-of-near-zero. Branches with jax.lax.cond; JIT-safe.

    Verified against `mujoco.mju_mat2Quat` for ±90° rotations around each axis.
    """
    trace = m[0, 0] + m[1, 1] + m[2, 2]

    def trace_pos(_):
        s = jp.sqrt(trace + 1.0) * 2
        return jp.array([0.25 * s,
                         (m[2, 1] - m[1, 2]) / s,
                         (m[0, 2] - m[2, 0]) / s,
                         (m[1, 0] - m[0, 1]) / s])

    def case_xx(_):
        s = jp.sqrt(1.0 + m[0, 0] - m[1, 1] - m[2, 2]) * 2
        return jp.array([(m[2, 1] - m[1, 2]) / s,
                         0.25 * s,
                         (m[0, 1] + m[1, 0]) / s,
                         (m[0, 2] + m[2, 0]) / s])

    def case_yy(_):
        s = jp.sqrt(1.0 + m[1, 1] - m[0, 0] - m[2, 2]) * 2
        return jp.array([(m[0, 2] - m[2, 0]) / s,
                         (m[0, 1] + m[1, 0]) / s,
                         0.25 * s,
                         (m[1, 2] + m[2, 1]) / s])

    def case_zz(_):
        s = jp.sqrt(1.0 + m[2, 2] - m[0, 0] - m[1, 1]) * 2
        return jp.array([(m[1, 0] - m[0, 1]) / s,
                         (m[0, 2] + m[2, 0]) / s,
                         (m[1, 2] + m[2, 1]) / s,
                         0.25 * s])

    # Branch on max diagonal element when trace ≤ 0.
    xx, yy, zz = m[0, 0], m[1, 1], m[2, 2]
    def trace_neg(_):
        return jax.lax.cond(
            (xx > yy) & (xx > zz),
            case_xx,
            lambda _: jax.lax.cond(yy > zz, case_yy, case_zz, operand=None),
            operand=None,
        )

    return jax.lax.cond(trace > 0, trace_pos, trace_neg, operand=None)


def _quat_finite_diff_angvel(q_prev: jp.ndarray, q_curr: jp.ndarray, dt: float) -> jp.ndarray:
    """Body-frame angular velocity from successive quaternions."""
    # ω ≈ 2 / dt * (q_curr * q_prev^*).xyz
    qpc = _quat_mul(q_curr, _quat_conj(q_prev))
    return 2.0 * qpc[1:] / dt


def _quat_mul(a: jp.ndarray, b: jp.ndarray) -> jp.ndarray:
    w1, x1, y1, z1 = a
    w2, x2, y2, z2 = b
    return jp.array([
        w1*w2 - x1*x2 - y1*y2 - z1*z2,
        w1*x2 + x1*w2 + y1*z2 - z1*y2,
        w1*y2 - x1*z2 + y1*w2 + z1*x2,
        w1*z2 + x1*y2 - y1*x2 + z1*w2,
    ])


def _quat_conj(q: jp.ndarray) -> jp.ndarray:
    return jp.array([q[0], -q[1], -q[2], -q[3]])
```

    def step(self, state: mjx_env.State, action: jp.ndarray) -> mjx_env.State:
        from jax_rl.envs.manipulation.factory.controller.action_chain import (
            apply_ema, denormalize, clip_to_bounds, reset_on_done,
        )

        info = state.info
        # Step 1: clear ema if just-reset (state.info auto-reset trap)
        done = state.done.astype(jp.bool_)
        ema_prev = info["actions"]
        ema_prev = jp.where(done, jp.zeros_like(ema_prev), ema_prev)
        # Step 2: apply EMA
        ema = apply_ema(action, ema_prev, ema_factor=self._config.ema_factor)
        # Step 3: denorm + clip (target reference frame is bolt pos)
        pos_delta, rot_delta = denormalize(
            ema,
            jp.array(self._config.pos_threshold),
            jp.array(self._config.rot_threshold),
            self._config.unidirectional_rot,
        )
        fingertip_pos = state.data.site_xpos[self._fingertip_site]
        target_pos = clip_to_bounds(
            fingertip_pos + pos_delta,
            info["fixed_pos"],
            jp.array(self._config.pos_action_bounds),
        )

        # PLACEHOLDER controller — zero torques. Phase 2 wires real OSC here.
        ctrl = jp.zeros(self.mjx_model.nu)

        # Inner substep loop @ sim_dt
        def inner_step(_, data):
            return mjx.step(self.mjx_model, data.replace(ctrl=ctrl))
        data = jax.lax.fori_loop(0, self._config.decimation, inner_step, state.data)

        # Update info
        prev_fingertip_pos  = state.data.site_xpos[self._fingertip_site]
        prev_fingertip_quat = self._fingertip_quat(state.data)
        new_info = {
            **info,
            "actions":      ema,
            "prev_actions": info["actions"],
            "prev_fingertip_pos":  prev_fingertip_pos,
            "prev_fingertip_quat": prev_fingertip_quat,
            "step_count":   info["step_count"] + 1,
        }

        # Compute reward
        from jax_rl.envs.manipulation.factory.reward import compute_reward
        target_held_pos = info["fixed_pos"]   # at success, peg sits inside hole
        target_held_quat = info["fixed_quat"]
        peg_z = data.xpos[self._peg_body_id, 2]
        hole_top_z = info["fixed_pos"][2] + self._config.asset_height
        reward = compute_reward(
            held_pos=data.xpos[self._peg_body_id],
            held_quat=data.xquat[self._peg_body_id],
            target_pos=target_held_pos,
            target_quat=target_held_quat,
            peg_z=peg_z,
            hole_top_z=hole_top_z,
            keypoint_coef_baseline=self._config.keypoint_coef_baseline,
            keypoint_coef_coarse=self._config.keypoint_coef_coarse,
            keypoint_coef_fine=self._config.keypoint_coef_fine,
            asset_height=self._config.asset_height,
            engage_threshold=self._config.engage_threshold,
            success_threshold=self._config.success_threshold,
            keypoint_scale=self._config.keypoint_scale,
        )

        # Done: TimeLimit at episode_length
        truncated = new_info["step_count"] >= self._config.episode_length
        # No early-terminate condition for PegInsert (peg can't fall — welded).
        terminated = jp.array(False)
        done = (truncated | terminated).astype(jp.float32)

        # truncation flag (Brax convention) for SAC loss masking
        new_info["truncation"] = truncated.astype(jp.float32)

        # compute_obs threads rng through info["rng"] (pattern from
        # go2_warp_joystick.py:398-403). Do NOT mint a fresh PRNGKey every step
        # — that gives identical noise samples and breaks DR/noise diversity.
        obs, new_info["rng"] = compute_obs(
            self._obs_groups,
            noise_level=0.0,
            rng=new_info["rng"],
            data=data, info=new_info,
        )

        return mjx_env.State(
            data=data, obs=obs, reward=reward, done=done, info=new_info,
        )

  - Task wrench:  F = Kp·err - Kd·vel
  - Task torque:  τ_task = J^T · Λ · wrench           where Λ = (J·M^-1·J^T)^-1
  - Nullspace:    τ_null = (I - J^T·Λ·J·M^-1) · M · (kp_null·(q_def - q) - kd_null·qdot)
  - Final:        τ = clamp(τ_task + τ_null, ±torque_limit)

fp64 escape hatch: wrap controller in `with jax.experimental.enable_x64()` if
Lambda inverse goes ill-conditioned near singular configs.
"""
from typing import Tuple

import jax
import jax.numpy as jp
from mujoco import mjx


def compute_site_jacobian(model, data, site_id: int) -> jp.ndarray:
    """Return (6, nv) geometric Jacobian of a site, world frame.

    Rows 0:3 = linear, rows 3:6 = angular. Matches MuJoCo's mj_jacSite.

    Implementation notes (verified against mujoco.mjx._src.support.jac):
    - `mjx.jac(model, data, point, body_id)` returns a pair `(jacp, jacr)`
      each of shape `(NV, 3)` — note the order is (NV, 3), NOT (3, NV).
    - `site_bodyid` lives on Model, not Data. Use `model.site_bodyid[site_id]`.
    - Transpose each piece to (3, nv) then concat along axis=0 → (6, nv).
    """
    jacp, jacr = mjx.jac(
        model,
        data,
        data.site_xpos[site_id],     # point in world frame
        model.site_bodyid[site_id],  # parent body id (Model field, NOT Data)
    )
    # jacp, jacr are (NV, 3). Transpose to (3, NV) and concat to (6, NV).
    return jp.concatenate([jacp.T, jacr.T], axis=0)
```

```python
def compute_pose_error(
    cur_pos: jp.ndarray, cur_quat: jp.ndarray,
    tgt_pos: jp.ndarray, tgt_quat: jp.ndarray,
) -> Tuple[jp.ndarray, jp.ndarray]:
    """Position error (linear) + rotation error (axis-angle) for OSC."""
    pos_err = tgt_pos - cur_pos
    # Rotation error = log_quat(target * conj(current)).
    err_quat = _quat_mul(tgt_quat, _quat_conj(cur_quat))
    # Convert to axis-angle (small-angle stable):
    sin_half = jp.linalg.norm(err_quat[1:])
    cos_half = err_quat[0]
    angle = 2.0 * jp.arctan2(sin_half, cos_half)
    # Avoid division by zero when sin_half is tiny.
    axis = jp.where(sin_half > 1e-8,
                    err_quat[1:] / jp.maximum(sin_half, 1e-8),
                    jp.zeros(3))
    rot_err = axis * angle
    return pos_err, rot_err


def _quat_mul(a, b):
    w1, x1, y1, z1 = a
    w2, x2, y2, z2 = b
    return jp.array([
        w1*w2 - x1*x2 - y1*y2 - z1*z2,
        w1*x2 + x1*w2 + y1*z2 - z1*y2,
        w1*y2 - x1*z2 + y1*w2 + z1*x2,
        w1*z2 + x1*y2 - y1*x2 + z1*w2,
    ])


def _quat_conj(q):
    return jp.array([q[0], -q[1], -q[2], -q[3]])
```

```python
def compute_osc_torque(
    model, data,
    target_pos: jp.ndarray, target_quat: jp.ndarray,
    site_id: int,
    arm_dof_ids: jp.ndarray,           # (7,) joint indices for arm
    kp_task: jp.ndarray,               # (6,)
    kd_task: jp.ndarray,               # (6,)
    q_default: jp.ndarray,             # (7,)
    kp_null: float, kd_null: float,
    torque_limit: float = 100.0,
) -> jp.ndarray:
    """Return (nu,) joint torques implementing Khatib OSC."""
    # 1. FK + Jacobian
    fingertip_pos = data.site_xpos[site_id]
    fingertip_mat = data.site_xmat[site_id].reshape(3, 3)
    from jax_rl.envs.manipulation.factory.factory_peg_insert import _mat_to_quat
    fingertip_quat = _mat_to_quat(fingertip_mat)

    J_full = compute_site_jacobian(model, data, site_id)        # (6, nv)
    J_arm = J_full[:, arm_dof_ids]                              # (6, 7)
    qvel_arm = data.qvel[arm_dof_ids]                           # (7,)
    fingertip_vel = J_arm @ qvel_arm                             # (6,)

    # 2. Pose error
    pos_err, rot_err = compute_pose_error(
        fingertip_pos, fingertip_quat, target_pos, target_quat
    )

    # 3. Task wrench
    wrench = jp.concatenate([
        kp_task[:3] * pos_err - kd_task[:3] * fingertip_vel[:3],
        kp_task[3:] * rot_err - kd_task[3:] * fingertip_vel[3:],
    ])

    # 4. Mass matrix slice
    # NOTE: mjx.full_m (lowercase m) — `mjx.full_M` does NOT exist.
    # Slice both rows and cols by arm_dof_ids to get the 7x7 arm submatrix.
    M_full = mjx.full_m(model, data)
    M_arm = M_full[jp.ix_(arm_dof_ids, arm_dof_ids)]            # (7, 7)
    M_inv = jp.linalg.inv(M_arm)

    # 5. DC nullspace
    Lambda = jp.linalg.inv(J_arm @ M_inv @ J_arm.T)             # (6, 6)
    J_eef_inv = Lambda @ J_arm @ M_inv                          # (6, 7)
    null_proj = jp.eye(7) - J_arm.T @ J_eef_inv                 # (7, 7)
    qpos_arm = data.qpos[arm_dof_ids]
    tau_null_raw = M_arm @ (kp_null * (q_default - qpos_arm) - kd_null * qvel_arm)
    tau_null = null_proj @ tau_null_raw

    # 6. Task torque
    tau_task = J_arm.T @ Lambda @ wrench                        # (7,)

    # 7. Combine + clamp; pad to (nu,) — gripper joints get zero.
    tau_arm = jp.clip(tau_task + tau_null, -torque_limit, torque_limit)
    nu = model.nu
    tau = jp.zeros(nu)
    # Map arm_dof_ids to actuator indices (assuming arm_jnt id == actuator id for 7-DOF panda).
    tau = tau.at[arm_dof_ids].set(tau_arm)
    return tau
```
