# jax_rl/networks/heads/flow.py
"""Flow vector field head for FPO.

Implements the conditional flow matching policy from
McAllister et al. (2025), arXiv:2507.21053.
"""

from __future__ import annotations
import jax
import jax.numpy as jnp


def embed_timestep(t: jax.Array, embed_dim: int) -> jax.Array:
    """Sinusoidal embedding of flow timestep.

    Frequencies are 2^i for i in [0, embed_dim/2). Output is concat([cos, sin]).

    Args:
        t: (..., 1) timestep in [0, 1]
        embed_dim: must be even

    Returns:
        (..., embed_dim) embedding
    """
    assert embed_dim % 2 == 0
    assert t.shape[-1] == 1
    freqs = 2 ** jnp.arange(embed_dim // 2)
    scaled = t * freqs  # (..., embed_dim/2)
    return jnp.concatenate([jnp.cos(scaled), jnp.sin(scaled)], axis=-1)
```

# jax_rl/networks/heads/flow.py (append)
from typing import Literal

OutputMode = Literal["u", "u_but_supervise_as_eps"]


def compute_cfm_loss_terms(
    action: jax.Array,        # (..., action_dim)
    eps: jax.Array,           # (..., n_samples, action_dim)
    t: jax.Array,             # (..., n_samples, 1)
    velocity_pred: jax.Array, # (..., n_samples, action_dim)
    output_mode: OutputMode,
) -> jax.Array:
    """Per-sample CFM loss term, mean-squared over action dims.

    The interpolated point is x_t = t*eps + (1-t)*action.
    Ground-truth velocity (rectified flow) is u = eps - action.

    Returns: (..., n_samples) loss values.
    """
    a_expanded = action[..., None, :]            # (..., 1, action_dim)
    if output_mode == "u":
        velocity_gt = eps - a_expanded
        return jnp.mean((velocity_pred - velocity_gt) ** 2, axis=-1)
    elif output_mode == "u_but_supervise_as_eps":
        x_t = t * eps + (1.0 - t) * a_expanded
        x0_pred = x_t - t * velocity_pred         # x0 = action-side
        x1_pred = x0_pred + velocity_pred         # x1 = noise-side = eps target
        return jnp.mean((eps - x1_pred) ** 2, axis=-1)
    else:
        raise ValueError(f"Unknown output_mode: {output_mode}")
```

# jax_rl/networks/heads/flow.py (append)
from typing import Callable


def euler_sample(
    velocity_fn: Callable[[jax.Array, jax.Array, jax.Array], jax.Array],
    x_init: jax.Array,    # (..., action_dim) — typically Gaussian noise
    flow_steps: int,
    obs: jax.Array,       # (..., obs_dim)
) -> jax.Array:
    """Euler integration of a flow vector field from t=1 (noise) to t=0 (action).

    velocity_fn(x_t, tau, obs) -> velocity, shape matching x_t.
    tau is a scalar (broadcast inside velocity_fn as needed).
    """
    schedule = jnp.linspace(1.0, 0.0, flow_steps + 1)
    t_current = schedule[:-1]
    t_next = schedule[1:]

    def step(x_t, inputs):
        tau, tau_next = inputs
        dt = tau_next - tau
        v = velocity_fn(x_t, tau, obs)
        return x_t + dt * v, None

    x_final, _ = jax.lax.scan(step, x_init, (t_current, t_next))
    return x_final
```

# jax_rl/networks/heads/flow.py (append)
import flax.linen as nn


class FlowHead(nn.Module):
    """Flow vector field head.

    Takes encoder output, current x_t, and current timestep tau.
    Returns predicted velocity (same shape as x_t).

    Conditioning is via concat: [h_obs, x_t, embed(tau)].
    """

    action_dim: int
    hidden_dim: int = 32
    n_layers: int = 4
    timestep_embed_dim: int = 8
    output_scale: float = 0.25
    activation: Callable = nn.silu

    @nn.compact
    def __call__(self, h_obs: jax.Array, x_t: jax.Array, tau: jax.Array) -> jax.Array:
        """
        h_obs: (..., encoder_dim)
        x_t:   (..., action_dim)
        tau:   (..., 1)
        Returns: (..., action_dim) velocity prediction
        """
        tau_embed = embed_timestep(tau, self.timestep_embed_dim)
        x = jnp.concatenate([h_obs, x_t, tau_embed], axis=-1)
        for _ in range(self.n_layers):
            x = nn.Dense(self.hidden_dim)(x)
            x = self.activation(x)
        x = nn.Dense(self.action_dim)(x)
        return x * self.output_scale
```

    cfg = FPOConfig(
        encoder=EncoderConfig(obs_dim=12, hidden_dim=64, n_layers=2),
        flow_head=FlowHeadConfig(action_dim=4, hidden_dim=32, n_layers=4),
        n_samples_per_action=4,

    fpo = FPO(config=cfg, obs_dim=12, action_dim=4, optimizer=optax.adam(3e-4))
    state = fpo.init(jax.random.PRNGKey(0))
    obs = jnp.ones((2, 12))
    action = jnp.zeros((2, 4))
    eps = jax.random.normal(jax.random.PRNGKey(1), (2, 4, 4))
    tau = jax.random.uniform(jax.random.PRNGKey(2), (2, 4, 1))
    grad_fn = jax.grad(
        lambda p: fpo._compute_cfm_loss(p, obs, action, eps, tau).mean()
    )
    grads = grad_fn(state.actor_params)
    leaves = jax.tree.leaves(grads)
    grad_norm = sum((l ** 2).sum() for l in leaves)

- [ ] **Step 2: Implement `_compute_cfm_loss`**

```python
# jax_rl/algos/fpo.py (inside FPO class)
def _compute_cfm_loss(
    self,
    actor_params,
    obs: jax.Array,             # (..., obs_dim)
    action: jax.Array,          # (..., action_dim)
    eps: jax.Array,             # (..., n_samples, action_dim)
    tau: jax.Array,             # (..., n_samples, 1)
) -> jax.Array:                 # (..., n_samples)
    """Per-sample CFM loss for stored actions."""
    n_samples = eps.shape[-2]
    # Compute x_t and broadcast obs over samples for vectorized vector-field call.
    x_t = tau * eps + (1.0 - tau) * action[..., None, :]
    obs_b = jnp.broadcast_to(
        obs[..., None, :],
        (*obs.shape[:-1], n_samples, obs.shape[-1]),
    )
    # Reshape to (-1, dim) for actor.apply, then unflatten.
    flat_obs = obs_b.reshape(-1, obs.shape[-1])
    flat_xt = x_t.reshape(-1, action.shape[-1])
    flat_tau = tau.reshape(-1, 1)
    flat_v = self.actor.apply(actor_params, flat_obs, flat_xt, flat_tau)
    v_pred = flat_v.reshape(*x_t.shape)
    return compute_cfm_loss_terms(action, eps, tau, v_pred, self.config.output_mode)
```
